create table students (
  students_student_id int identity not null,
  students_last_name varchar(50) not null,
  students_first_name varchar(50) not null,
  students_email varchar(50) not null,
  students_age int not null,
  students_grade int not null,
  students_teacher_id int not null,
  constraint pk_students_student_id primary key(students_student_id),
  constraint u_student_email unique(students_email)
)

create table teachers (
  teachers_teacher_id int identity not null,
  teachers_last_name varchar(50) not null,
  teachers_first_name varchar(50) not null,
  teachers_email varchar(50) not null,
  teachers_grade int not null,
  teachers_homeroom varchar(2) not null,

/* This is for bulk inserting our starting data to the database. */
create or alter procedure p_mint_upsert_students (
  @student_id int, 
  @last_name varchar(50), 
  @first_name varchar(50),
  @email_address varchar(50),
  @age int,
  @grade int,
  @teacher_id int
) as begin
  declare @check_student_id int
  select @check_student_id = students_student_id from students where students_student_id = @student_id
  if (@check_student_id > 0) 
    begin update students set 
      students_last_name = @last_name, students_first_name = @first_name,
      students_email= @email_address, students_age = @age,
      students_grade = @grade, students_teacher_id = @teacher_id
      where students_student_id = @student_id
    end
    else begin 
      insert into 
      students (students_student_id,students_last_name,students_first_name,students_email,students_age,students_grade,students_teacher_id)
      values (@student_id,@last_name,@first_name,@email_address,@age,@grade,@teacher_id)
    end
end

select 
    students.students_first_name
    ,students.students_last_name
    ,assessments.assessments_percent_correct
    ,assessments.assessments_raw_score
from students
    left join assessments on students.students_student_id = assessments.assessments_student_id
    left join math_skills on assessments.assessments_skill_id = math_skills.math_skills_skill_id
    left join teachers on students.students_teacher_id = teachers.teachers_teacher_id
where math_skills.math_skills_name = 'Add_Subtract_Regroup_100' and teachers.teachers_last_name = 'Shoots'
order by assessments_percent_correct desc

    ,students.students_last_name

    ,assessments.assessments_percent_correct
    ,assessments.assessments_raw_score

    left join math_skills on assessments.assessments_skill_id = math_skills.math_skills_skill_id

--Query Answer 3
select 
    students.students_first_name
    ,students.students_last_name
    ,math_skills.math_skills_name
    ,assessments.assessments_percent_correct
    ,assessments.assessments_raw_score
    ,AVG(practice_logs.practice_logs_practice_score) as avg_score
from students
    left join assessments on students.students_student_id = assessments.assessments_student_id
    left join math_skills on assessments.assessments_skill_id = math_skills.math_skills_skill_id
    left join teachers on students.students_teacher_id = teachers.teachers_teacher_id
    left join practice_logs on students.students_student_id = practice_logs.practice_logs_student_id
where teachers.teachers_last_name = 'Alinsonorin' and math_skills.math_skills_skill_id = 2
group by students.students_first_name
    ,students.students_last_name
    ,math_skills.math_skills_name
    ,assessments.assessments_percent_correct
    ,assessments.assessments_raw_score
order by assessments_percent_correct desc

--Query Answer 5
select 
    students.students_first_name
    ,students.students_last_name
    ,math_skills.math_skills_name
    ,AVG(practice_logs.practice_logs_practice_score) as avg_score
    ,count(1) as num_practices
from students
    left join assessments on students.students_student_id = assessments.assessments_student_id
    left join math_skills on assessments.assessments_skill_id = math_skills.math_skills_skill_id
    left join practice_logs on students.students_student_id = practice_logs.practice_logs_student_id
group by students.students_first_name
    ,students.students_last_name
    ,math_skills.math_skills_name
order by avg_score desc

--Query Answer 6
select 
    math_skills.math_skills_name
    ,avg(assessments.assessments_theorized_score) as avg_theory_score
    ,avg(assessments.assessments_raw_score) as avg_raw_score
    ,AVG(practice_logs.practice_logs_practice_score) as avg_pracice_score
from students
    left join assessments on students.students_student_id = assessments.assessments_student_id
    left join math_skills on assessments.assessments_skill_id = math_skills.math_skills_skill_id
    left join practice_logs on students.students_student_id = practice_logs.practice_logs_student_id
    left join teachers on teachers.teachers_teacher_id = students.students_teacher_id
where teachers_grade = 2  
group by students.students_first_name
    ,students.students_last_name
    ,math_skills.math_skills_name
order by avg_theory_score desc

--Query Answer 7
select students.students_first_name
  ,students.students_last_name
  ,students.students_grade
  ,avg(practice_logs.practice_logs_start_time - practice_logs.practice_logs_end_time) as avg_practice_time
  ,max(practice_logs.practice_logs_start_time - practice_logs.practice_logs_end_time) as longest_session
from students
    left join practice_logs on students.students_student_id = practice_logs.practice_logs_student_id
group by
  students.students_last_name, students.students_grade
order by
  4 desc
